'use client'

import { useState, useEffect } from 'react'
import { useSession } from 'next-auth/react'
import {
  Box,
  Button,
  Typography,
  Paper,
  Grid,
  MenuItem,
  Stack,
  Divider,
  Snackbar,
  Alert,
  CircularProgress,
  Chip,
  Switch,
  IconButton,
  LinearProgress
} from '@mui/material'

import {
  Person2Rounded,
  EditOutlined,
  InsertDriveFileOutlined,
  CloudUploadOutlined,
  DeleteOutlined,
  AutoAwesomeOutlined,
  Settings
} from '@mui/icons-material'

import {
  getPartnerProductsAPI,
  getAllAddCasesApi,
  getpartnerproduct,
  getAllServicesApi,
  getAllLocationsAPI,
  uploadImageApi,
  uploadMultiImageApi,
  postJobApi
} from '@/services/apiService'
import DataTable from '@/app/common/dataTable'
import Modal from '../../components/modal'
import CustomTextField from '@/@core/components/mui/TextField'

export default function InitiateCases() {
  const { data: session } = useSession()
  const [loading, setLoading] = useState(false)
  const [cases, setCases] = useState([])
  const [openAddModal, setOpenAddModal] = useState(false)
  const [openViewModal, setOpenViewModal] = useState(false)
  const [openExcelModal, setOpenExcelModal] = useState(false)
  const [selectedCase, setSelectedCase] = useState(null)
  const [partnerName, setPartnerName] = useState(null)
  const [serviceId, setServiceId] = useState(null)
  const [requestId, setRequestId] = useState(null)
  const [initId, setInitId] = useState(null)
  const [partnerProducts, setPartnerProducts] = useState([])
  const [formVisibility, setFormVisibility] = useState({})
  const [productFields, setProductFields] = useState({})
  const [uploadingFiles, setUploadingFiles] = useState({})
  const [fieldErrors, setFieldErrors] = useState({})
  const [aiProcessing, setAiProcessing] = useState({})
  const [aiButtonVisible, setAiButtonVisible] = useState({})
  const [aiResponseData, setAiResponseData] = useState({})
  const [locations, setLocations] = useState([])
  const [services, setServices] = useState([])
  const [excelData, setExcelData] = useState({
    partnerId: '',
    reportType: '',
    documents: null
  })
  const [snackbar, setSnackbar] = useState({
    open: false,
    message: '',
    severity: 'success'
  })

  // Fetch cases on component mount
  useEffect(() => {
    fetchCases()
  }, [])

  // Fetch services on component mount
  useEffect(() => {
    fetchServices()
  }, [])

  // Initialize form visibility and fields when partner products change
  useEffect(() => {
    const initialVisibility = {}
    const initialData = {}

    partnerProducts.forEach(product => {
      initialVisibility[product._id] = false
      initialData[product._id] =
        product?.initFields?.fields?.reduce((acc, field) => {
          acc[field.fieldName] = ''
          return acc
        }, {}) || {}
    })

    setFormVisibility(initialVisibility)
    setProductFields(initialData)
  }, [partnerProducts])

  // Fetch locations when partner products change
  useEffect(() => {
    if (partnerProducts.length > 0) {
      fetchLocations()
    }
  }, [partnerProducts])

  const fetchCases = async () => {
    try {
      setLoading(true)
      const data = await getAllAddCasesApi()

      if (data?.items) {
        setCases(
          data.items.map(item => ({
            _id: item._id,
            partnerId: item.partnerId?.fullName || 'N/A',
            fileNo: item.fileNo,
            requestData: item.requestData || {},
            serviceId: item.serviceId,
            customerName: item.customerName,
            fatherName: item.fatherName,
            contactNo: item.contactNo,
            address: item.address
          }))
        )
      }
    } catch (err) {
      console.error('Failed to fetch cases:', err)
      showSnackbar('Failed to fetch cases', 'error')
    } finally {
      setLoading(false)
    }
  }

  const fetchServices = async () => {
    try {
      setLoading(true)
      const response = await getAllServicesApi()
      console.log('services', response)

      if (response?.items) {
        setServices(response.items)
      }
    } catch (err) {
      console.error('Failed to fetch services:', err)
    } finally {
      setLoading(false)
    }
  }

  const fetchLocations = async () => {
    try {
      const data = await getAllLocationsAPI()

      if (data?.items) {
        setLocations(data.items)
      }
    } catch (err) {
      console.error('Failed to fetch locations:', err)
    }
  }

  const handleOpenAddModal = async row => {
    setOpenAddModal(true)
    const requestId = row.requestData._id
    const initId = row._id
    setRequestId(requestId)
    setPartnerName(row.partnerId)
    setInitId(initId)

    if (serviceId) {
      try {
        const response = await getPartnerProductsAPI(requestId, serviceId, initId)
        if (response && response.items) {
          setPartnerProducts(response.items)
        }
      } catch (error) {
        console.error('Error fetching partner products:', error)
        showSnackbar('Failed to fetch partner products', 'error')
      }
    }
  }

  const handleOpenViewModal = row => {
    setOpenViewModal(true)
    setSelectedCase(row)
  }

  const handleOpenExcelModal = () => {
    setOpenExcelModal(true)
  }

  const handleCloseAddModal = () => {
    setOpenAddModal(false)
    resetForm()
  }

  const handleCloseViewModal = () => {
    setOpenViewModal(false)
    setSelectedCase(null)
  }

  const handleCloseExcelModal = () => {
    setOpenExcelModal(false)
    setExcelData({
      partnerId: '',
      reportType: '',
      documents: null
    })
  }

  const handleServiceChange = async e => {
    const newServiceId = e.target.value
    setServiceId(newServiceId)

    if (openAddModal && requestId && initId && newServiceId) {
      try {
        const response = await getpartnerproduct(requestId, newServiceId, initId)
        console.log('service product', response)

        if (response && response.items) {
          setPartnerProducts(response.items)
        }
      } catch (error) {
        console.error('Error fetching partner products:', error)
        showSnackbar('Failed to fetch partner products', 'error')
      }
    }
  }

  const handleToggleProduct = productId => {
    setFormVisibility(prev => ({
      ...prev,
      [productId]: !prev[productId]
    }))
  }

  const handleProductFieldChange = (productId, fieldName, value) => {
    setProductFields(prev => ({
      ...prev,
      [productId]: {
        ...prev[productId],
        [fieldName]: value
      }
    }))

    // Clear error when user starts typing
    if (fieldErrors[`${productId}_${fieldName}`]) {
      setFieldErrors(prev => ({
        ...prev,
        [`${productId}_${fieldName}`]: undefined
      }))
    }
  }

  const handleLocationChange = (productId, locationId) => {
    setProductFields(prev => ({
      ...prev,
      [productId]: {
        ...prev[productId],
        locationId: locationId
      }
    }))

    // Clear error when user selects location
    if (fieldErrors[`${productId}_locationId`]) {
      setFieldErrors(prev => ({
        ...prev,
        [`${productId}_locationId`]: undefined
      }))
    }
  }

  const handleFileUpload = async (file, productId, fieldName) => {
    if (!file) return

    setUploadingFiles(prev => ({
      ...prev,
      [`${productId}_${fieldName}`]: true
    }))

    try {
      const response = await uploadImageApi(file)

      if (response.status && response.items?.fileUrl) {
        handleProductFieldChange(productId, fieldName, response.items.fileUrl)
        handleProductFieldChange(productId, `${fieldName}_filename`, file.name)

        // Make AI button visible for this product
        setAiButtonVisible(prev => ({
          ...prev,
          [productId]: true
        }))
      } else {
        console.error('File upload failed:', response)
        showSnackbar('File upload failed', 'error')
      }
    } catch (error) {
      console.error('Error uploading file:', error)
      showSnackbar('Error uploading file', 'error')
    } finally {
      setUploadingFiles(prev => ({
        ...prev,
        [`${productId}_${fieldName}`]: false
      }))
    }
  }

  const handleMultiFileUpload = async (files, productId, fieldName) => {
    if (!files || files.length === 0) return

    // Initialize upload progress indicators for each file
    const uploadItems = files.map(file => ({
      filename: file.name,
      size: file.size,
      progress: 0,
      status: 'uploading'
    }))

    setUploadingFiles(prev => ({
      ...prev,
      [`${productId}_${fieldName}_items`]: uploadItems
    }))

    try {
      const formData = new FormData()
      files.forEach(file => {
        formData.append('images', file)
      })

      const response = await uploadMultiImageApi(formData)

      if (response.status && response.items?.files) {
        // Update progress to 100% for all files
        setUploadingFiles(prev => ({
          ...prev,
          [`${productId}_${fieldName}_items`]: prev[`${productId}_${fieldName}_items`].map(item => ({
            ...item,
            progress: 100,
            status: 'completed'
          }))
        }))

        // Get existing files if any
        const existingFiles = Array.isArray(productFields[productId]?.[fieldName])
          ? [...productFields[productId][fieldName]]
          : []

        // Combine with new files
        const updatedFiles = [...existingFiles, ...response.items.files].flat()

        // Store the file URLs array
        handleProductFieldChange(productId, fieldName, updatedFiles)

        // Get existing filenames if any
        const existingFilenames = Array.isArray(productFields[productId]?.[`${fieldName}_filenames`])
          ? [...productFields[productId][`${fieldName}_filenames`]]
          : []

        // Store filenames for display purposes - combine with existing
        const filenames = [...existingFilenames, ...files.map(file => file.name)]
        handleProductFieldChange(productId, `${fieldName}_filenames`, filenames)

        // Make AI button visible for this product
        setAiButtonVisible(prev => ({
          ...prev,
          [productId]: true
        }))

        // Clear the temporary uploading files state after a delay
        setTimeout(() => {
          setUploadingFiles(prev => {
            const newState = { ...prev }
            delete newState[`${productId}_${fieldName}_items`]
            return newState
          })
        }, 2000)
      } else {
        console.error('Multi-file upload failed:', response)
        showSnackbar('Multi-file upload failed', 'error')

        // Update status to show error
        setUploadingFiles(prev => ({
          ...prev,
          [`${productId}_${fieldName}_items`]: prev[`${productId}_${fieldName}_items`].map(item => ({
            ...item,
            status: 'error'
          }))
        }))
      }
    } catch (error) {
      console.error('Error uploading multiple files:', error)
      showSnackbar('Error uploading multiple files', 'error')

      // Update status to show error
      setUploadingFiles(prev => ({
        ...prev,
        [`${productId}_${fieldName}_items`]:
          prev[`${productId}_${fieldName}_items`]?.map(item => ({
            ...item,
            status: 'error'
          })) || []
      }))
    }
  }

  const handleRemoveFile = (productId, fieldName, index) => {
    // For single file upload
    if (index === undefined) {
      handleProductFieldChange(productId, fieldName, '')
      handleProductFieldChange(productId, `${fieldName}_filename`, '')
      return
    }

    // For multi-file upload
    const newFiles = [...productFields[productId][fieldName]]
    newFiles.splice(index, 1)

    // Update the files array
    handleProductFieldChange(productId, fieldName, newFiles)

    // Update filenames array if exists
    if (productFields[productId]?.[`${fieldName}_filenames`]) {
      const newFilenames = [...productFields[productId][`${fieldName}_filenames`]]
      newFilenames.splice(index, 1)
      handleProductFieldChange(productId, `${fieldName}_filenames`, newFilenames)
    }
  }

  const runAIDataExtraction = async productId => {
    setAiProcessing(prev => ({
      ...prev,
      [productId]: true
    }))

    try {
      // Find the current product
      const currentProduct = partnerProducts.find(p => p._id === productId)

      if (!currentProduct) {
        console.error('Product information not found')
        return
      }

      // Get all file URLs for this product
      const fileUrls = []

      if (currentProduct?.initFields?.fields) {
        currentProduct.initFields.fields.forEach(field => {
          if (
            field.dataType !== 'string' &&
            field.dataType !== 'textarea' &&
            productFields[productId]?.[field.fieldName]
          ) {
            fileUrls.push(productFields[productId][field.fieldName])
          }
        })
      }

      // Only proceed if we have files
      if (fileUrls.length === 0) {
        console.warn('No files found for AI processing')
        showSnackbar('No files found for AI processing', 'warning')
        return
      }

      const flatFileUrls = fileUrls.flat()

      const payload = {
        reqId: currentProduct.userProductId,
        userProductId: currentProduct.userProductId,
        doc: flatFileUrls
      }

      // const response = await processWithAI(payload)

      if (response.status && response.items) {
        // Update the form fields with the extracted data
        const extractedData = response.items

        // Create updated product fields
        const updatedFields = {
          ...productFields[productId]
        }

        // Loop through all the extracted fields and update corresponding form fields
        Object.keys(extractedData).forEach(fieldName => {
          // Only update if the field has a value
          if (extractedData[fieldName] !== null) {
            updatedFields[fieldName] = extractedData[fieldName]
          }
        })

        // Update the product fields state with new data
        setProductFields(prev => ({
          ...prev,
          [productId]: updatedFields
        }))

        // Store the AI response data for this product
        setAiResponseData(prev => ({
          ...prev,
          [productId]: {
            reqId: currentProduct.userProductId,
            userProductId: currentProduct.userProductId,
            doc: fileUrls
          }
        }))

        showSnackbar('AI data extraction completed successfully', 'success')
      } else {
        const errorMessage = response.message || 'Failed to extract data from documents'
        console.error('AI Extraction Error:', errorMessage)
        showSnackbar(errorMessage, 'error')
      }
    } catch (error) {
      console.error('Error in AI data extraction:', error)
      showSnackbar('Error processing documents', 'error')
    } finally {
      setAiProcessing(prev => ({
        ...prev,
        [productId]: false
      }))
    }
  }

  const validateForm = () => {
    const errors = {}
    let isValid = true
    let hasAnyVisibleProduct = false
    const errorMessages = []

    // Validate each product's fields
    Object.entries(formVisibility).forEach(([productId, isVisible]) => {
      // Only validate visible/selected products
      if (!isVisible) return

      hasAnyVisibleProduct = true

      // Find the product to access its field definitions
      const product = partnerProducts.find(p => p._id === productId)
      if (!product) return

      // Check if product exists in productFields
      if (!productFields[productId]) {
        errors[`${productId}_general`] = 'Product data is missing'
        errorMessages.push(`${product.productName}: Product data is missing`)
        isValid = false
        return
      }

      const fields = productFields[productId] || {}

      // Validate location field (required for all products)
      if (!fields.locationId) {
        errors[`${productId}_locationId`] = 'Location is required'
        errorMessages.push(`${product.productName}: Location is required`)
        isValid = false
      }

      // Validate each custom field based on the product's initFields
      product?.initFields?.fields?.forEach(field => {
        const fieldValue = fields[field.fieldName]

        // Only validate fields with dataType "string"
        if (field.dataType === 'string') {
          // Check if field is required (all string fields are required)
          if (fieldValue === undefined || fieldValue === null || fieldValue === '') {
            errors[`${productId}_${field.fieldName}`] = `${field.fieldName} is required`
            errorMessages.push(`${product.productName}: ${field.fieldName} is required`)
            isValid = false
          }
        }
      })
    })

    // Check if no products are selected/visible
    if (!hasAnyVisibleProduct) {
      errorMessages.push('Please enable at least one product')
      isValid = false
    }

    // Set field errors to update UI
    setFieldErrors(errors)

    // Show validation errors in snackbar if any
    if (errorMessages.length > 0) {
      showSnackbar(errorMessages[0], 'error')
    }

    return isValid
  }

  const handleFormSubmit = async e => {
    e.preventDefault()

    // Run validation first
    const isValid = validateForm()
    if (!isValid) return

    // Prepare data for submission with correct structure
    // Only include products that are visible/toggled on
    const jobProductList = Object.entries(formVisibility)
      .filter(([productId, isVisible]) => isVisible)
      .map(([productId]) => {
        // Find the product to get its userProductId
        const product = partnerProducts[0]?.productForm?.find(p => p._id === productId)
        const fields = productFields[productId] || {}

        if (!product) return null

        // Extract field values in order
        const initValues =
          product?.initFields?.fields?.map(field => {
            return fields[field.fieldName] || ''
          }) || []

        // Create base product entry
        const productEntry = {
          userProductId: product.userProductId,
          locationId: fields.locationId,
          initValues
        }

        // If we have AI data for this product, include it in the payload
        if (aiResponseData && aiResponseData[productId]) {
          productEntry.aiBody = {
            reqId: aiResponseData[productId].reqId,
            userProductId: product.userProductId,
            doc: aiResponseData[productId].doc || []
          }
        }

        return productEntry
      })
      .filter(Boolean) // Remove any null entries

    // Double-check that we have products after filtering
    if (jobProductList.length === 0) {
      showSnackbar('No products selected. Please enable at least one product.', 'error')
      return
    }

    const payload = {
      partnerId: requestId,
      reportType: 'separated', // You might want to make this configurable
      jobProductList
    }
    console.log('form payload', payload)

    try {
      const data = await postJobApi(payload)

      if (data.status) {
        showSnackbar("Case successfully submitted", "success")
        handleCloseAddModal()
        fetchCases() // Refresh case list after submission
      } else {
        showSnackbar(data.message || "Failed to submit case", "error")
      }
    } catch (error) {
      console.error("Error submitting case:", error)
      showSnackbar("Failed to submit case", "error")
    }
  }

  const handleExcelSubmit = async e => {
    e.preventDefault()

    if (!excelData.partnerId || !excelData.reportType || !excelData.documents) {
      showSnackbar('Please fill all fields and upload a file', 'error')
      return
    }

    const formData = new FormData()
    formData.append('partnerId', excelData.partnerId)
    formData.append('reportType', excelData.reportType)
    formData.append('file', excelData.documents)

    try {
      // Replace with your actual API call
      const response = await fetch('/api/excel-upload', {
        method: 'POST',
        body: formData
      })

      const data = await response.json()

      if (data.status) {
        showSnackbar('Excel data submitted successfully', 'success')
        handleCloseExcelModal()
        fetchCases() // Refresh case list after submission
      } else {
        showSnackbar(data.message || 'Excel data submission failed', 'error')
      }
    } catch (error) {
      console.error('Error submitting Excel data:', error)
      showSnackbar('Failed to submit Excel data', 'error')
    }
  }

  const handleDownloadSampleExcel = async partnerId => {
    if (!partnerId) {
      showSnackbar('Please select a partner first', 'warning')
      return
    }

    try {
      // Replace with your actual API call
      // const response = await dow

      if (response.ok) {
        const blob = await response.blob()
        const url = window.URL.createObjectURL(blob)
        const a = document.createElement('a')
        a.href = url
        a.download = 'sample-excel-template.xlsx'
        document.body.appendChild(a)
        a.click()
        window.URL.revokeObjectURL(url)
        document.body.removeChild(a)
      } else {
        showSnackbar('Sample template not available', 'error')
      }
    } catch (error) {
      console.error('Error downloading sample Excel:', error)
      showSnackbar('Failed to download sample Excel', 'error')
    }
  }

  const resetForm = () => {
    setFormVisibility({})
    setProductFields({})
    setFieldErrors({})
    setAiResponseData({})
    setAiButtonVisible({})
    setAiProcessing({})
    setUploadingFiles({})
  }

  const showSnackbar = (message, severity = 'success') => {
    setSnackbar({
      open: true,
      message,
      severity
    })

    // Auto-hide after 5 seconds
    setTimeout(() => {
      setSnackbar(prev => ({ ...prev, open: false }))
    }, 5000)
  }

  // Helper functions for file fields
  const getFileTypeFieldsCount = productId => {
    const currentProduct = partnerProducts.find(p => p._id === productId)

    if (!currentProduct?.initFields?.fields) {
      return 0
    }

    return currentProduct.initFields.fields.filter(field => field.dataType !== 'string').length
  }

  const getPopulatedFileFieldsCount = productId => {
    const currentProduct = partnerProducts.find(p => p._id === productId)

    if (!currentProduct?.initFields?.fields) {
      return 0
    }

    let count = 0

    currentProduct.initFields.fields.forEach(field => {
      if (field.dataType !== 'string' && productFields[productId]?.[field.fieldName]) {
        count++
      }
    })

    return count
  }

  const columns = [
    {
      field: 'partnerId',
      headerName: 'Partner Name',
      flex: 0.5,
      minWidth: 120
    },
    {
      field: 'fileNo',
      headerName: 'File Number',
      flex: 0.5,
      minWidth: 100
    },
    {
      field: 'customerName',
      headerName: 'Customer Name',
      flex: 1,
      minWidth: 150
    },
    {
      field: 'fatherName',
      headerName: "Father's Name",
      flex: 1,
      minWidth: 150
    },
    {
      field: 'contactNo',
      headerName: 'Contact Number',
      flex: 0.75,
      minWidth: 130
    },
    {
      field: 'address',
      headerName: 'Address',
      flex: 1.5,
      minWidth: 200
    },
    {
      field: 'action',
      headerName: 'ACTION',
      flex: 0.75,
      minWidth: 120,
      renderCell: params => (
        <>
          <IconButton color='primary' onClick={() => handleOpenViewModal(params.row)}>
            <EditOutlined />
          </IconButton>
        </>
      )
    }
  ]

  return (
    <>
      <Paper sx={{ p: 5 }}>
        <Box sx={{ display: 'flex', justifyContent: 'space-between', my: 4 }}>
          <Typography sx={{ fontSize: '20px', fontWeight: 'bold' }}>Initiate Cases</Typography>
          <Box>
            <Button
              size='large'
              variant='contained'
              color='primary'
              sx={{ marginRight: 2 }}
              onClick={handleOpenExcelModal}
            >
              Upload Excel
            </Button>
          </Box>
        </Box>

        {loading ? (
          <Box sx={{ display: 'flex', justifyContent: 'center', alignItems: 'center', height: '50vh' }}>
            <CircularProgress size={60} />
          </Box>
        ) : (
          <DataTable
            columns={columns}
            rows={cases}
            page={0}
            rowsPerPage={10}
            totalItems={cases.length}
            extraActions={row => (
              <Button
                variant='contained'
                color='success'
                onClick={() => handleOpenAddModal(row)}
                sx={{ marginRight: 2 }}
              >
                Add Cases
              </Button>
            )}
          />
        )}
      </Paper>

      {/* Add Case Modal */}
      <Modal
        open={openAddModal}
        handleClose={handleCloseAddModal}
        showButton={false}
        title='Add New Case'
        maxWidth='md'
      >
        <Box component='form' sx={{ p: { xs: 2, md: 3 } }}>
          {/* Partner and Service Selection */}
          <Box sx={{ bgcolor: '#f5f7fa', p: 3, borderRadius: 2, mb: 3 }}>
            {/* Partner Section */}
            <Box sx={{ mb: 4 }}>
              <Box sx={{ display: 'flex', alignItems: 'center', mb: 2 }}>
                <Box
                  sx={{
                    width: 32,
                    height: 32,
                    borderRadius: '50%',
                    bgcolor: 'rgba(79, 70, 229, 0.1)',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    mr: 2
                  }}
                >
                  <Person2Rounded fontSize='small' sx={{ color: '#4F46E5' }} />
                </Box>
                <Typography variant='subtitle1' fontWeight={500}>
                  Partner
                </Typography>
              </Box>

              <CustomTextField
                fullWidth
                placeholder='Partner'
                value={partnerName || ''}
                disabled
                sx={{
                  '& .MuiOutlinedInput-root': {
                    borderRadius: 1.5,
                    height: 56,
                    bgcolor: 'white'
                  }
                }}
              />
            </Box>

            {/* Service Section */}
            <Box sx={{ mb: 4 }}>
              <Box sx={{ display: 'flex', alignItems: 'center', mb: 2 }}>

            <Box sx={{ p: 4, textAlign: 'center' }}>
              <Typography variant='h6' color='error'>
                No case selected.
              </Typography>
            </Box>
          )}
        </Box>
      </Modal>

      {/* Excel Upload Modal */}
      <Modal
        open={openExcelModal}
        handleClose={handleCloseExcelModal}
        title='Upload Excel'
        maxWidth='xs'
        fullWidth={true}
        showButton={false}
      >
        <Box component='form' onSubmit={handleExcelSubmit} sx={{ p: 3 }}>
          <Grid container spacing={2}>
            <Grid item xs={12}>
              <CustomTextField
                label='Select Partner'
                select
                fullWidth
                value={excelData.partnerId || ''}
                onChange={e => setExcelData({ ...excelData, partnerId: e.target.value })}
                sx={{ mb: 2 }}
              >
                {/* Replace with your partner data */}
                <MenuItem value='partner1'>Partner 1</MenuItem>
                <MenuItem value='partner2'>Partner 2</MenuItem>
              </CustomTextField>
            </Grid>
            <Grid item xs={12}>
              <CustomTextField
                label='Select Report Type'
                select
                fullWidth
                value={excelData.reportType || ''}
                onChange={e => setExcelData({ ...excelData, reportType: e.target.value })}
                sx={{ mb: 2 }}
              >
                <MenuItem value='merged'>Merged</MenuItem>
                <MenuItem value='separated'>Separated</MenuItem>
              </CustomTextField>
            </Grid>
            <Grid item xs={12}>
              <Box
                sx={{
                  border: '2px dashed #6366F1',
                  borderRadius: 1,
                  p: 4,
                  textAlign: 'center',
                  backgroundColor: '#F5F7FF',
                  my: 2
                }}
              >
                <input
                  accept='.xlsx,.xls,.csv'
                  id='excel-file-upload'
                  type='file'
                  style={{ display: 'none' }}
                  onChange={e => {
                    const file = e.target.files[0]
                    if (file) {
                      const fileExt = file.name.split('.').pop().toLowerCase()
                      if (fileExt === 'xlsx' || fileExt === 'xls' || fileExt === 'csv') {
                        setExcelData({ ...excelData, documents: file })
                      } else {
                        e.target.value = ''
                        showSnackbar('Please select only Excel files (.xlsx or .xls)', 'error')
                      }
                    }
                  }}
                />

                {!excelData.documents ? (
                  <>
                    <Box sx={{ display: 'flex', justifyContent: 'center', mb: 1 }}>
                      <Box
                        sx={{
                          width: 40,
                          height: 40,
                          borderRadius: '50%',
                          backgroundColor: '#E0E7FF',
                          display: 'flex',
                          alignItems: 'center',
                          justifyContent: 'center',
                          color: '#6366F1'
                        }}
                      >
                        <CloudUploadOutlined />
                      </Box>
                    </Box>

                    <Typography variant='body2' sx={{ mb: 1 }}>
                      Drag your file to upload
                    </Typography>

                    <Typography variant='body2' color='text.secondary' sx={{ mb: 2 }}>
                      OR
                    </Typography>

                    <label htmlFor='excel-file-upload'>
                      <Button
                        variant='outlined'
                        component='span'
                        color='primary'
                        size='small'
                        sx={{
                          borderRadius: 2,
                          px: 2,
                          py: 0.5,
                          textTransform: 'none',
                          borderColor: '#6366F1',
                          color: '#6366F1'
                        }}
                      >
                        Browse files
                      </Button>
                    </label>
                  </>
                ) : (
                  <Box>
                    <Typography variant='body2' sx={{ mb: 1 }}>
                      File selected: <strong>{excelData.documents.name}</strong>
                    </Typography>
                    <Button
                      variant='outlined'
                      color='error'
                      size='small'
                      onClick={() => setExcelData({ ...excelData, documents: null })}
                      sx={{ mt: 1 }}
                    >
                      Remove
                    </Button>
                  </Box>
                )}
              </Box>
            </Grid>
            <Grid item xs={12}>
              <Typography variant='caption' color='text.secondary'>
                {excelData.partnerId
                  ? 'Please ensure the Excel file is in the correct format.'
                  : 'Please select a partner to enable the download Excel.'}
              </Typography>
              <Button
                variant='outlined'
                color='inherit'
                onClick={() => (excelData.partnerId ? handleDownloadSampleExcel(excelData.partnerId) : null)}
                disabled={!excelData.partnerId}
                sx={{ mt: 2, mr: 2 }}
              >
                Download Sample Excel
              </Button>
            </Grid>
            <Grid item xs={12} sx={{ display: 'flex', justifyContent: 'flex-end', mt: 2 }}>
              <Button variant='text' color='inherit' onClick={handleCloseExcelModal} sx={{ mr: 2 }}>
                Cancel
              </Button>
              <Button
                variant='contained'
                type='submit'
                disabled={!excelData.partnerId || !excelData.reportType || !excelData.documents}
                sx={{
                  bgcolor: '#4F46E5',
                  '&:hover': {
                    bgcolor: '#4338CA'
                  }
                }}
              >
                Upload
              </Button>
            </Grid>
          </Grid>
        </Box>
      </Modal>

      {/* Snackbar for notifications */}
      <Snackbar
        open={snackbar.open}
        autoHideDuration={6000}
        onClose={() => setSnackbar({ ...snackbar, open: false })}
        anchorOrigin={{ vertical: 'top', horizontal: 'center' }}
      >
        <Alert
          onClose={() => setSnackbar({ ...snackbar, open: false })}
          variant='filled'
          severity={snackbar.severity}
          sx={{ width: '100%' }}
        >
          {snackbar.message}
        </Alert>
      </Snackbar>
    </>
  )
}

    if (serviceId) {

        showSnackbar("Failed to fetch partner products", "error")

    setOpenViewModal(true)

  }

  const handleToggleProduct = (productId) => {
    setFormVisibility((prev) => ({
      ...prev,
      [productId]: !prev[productId],
    }))
  }

  const handleProductFieldChange = (productId, fieldName, value) => {
    setProductFields((prev) => ({
      ...prev,
      [productId]: {
        ...prev[productId],
        [fieldName]: value,
      },
    }))

    // Clear error when user starts typing
    if (fieldErrors[`${productId}_${fieldName}`]) {
      setFieldErrors((prev) => ({
        ...prev,
        [`${productId}_${fieldName}`]: undefined,
      }))
    }
  }

  const handleLocationChange = (productId, locationId) => {
    setProductFields((prev) => ({
      ...prev,
      [productId]: {
        ...prev[productId],
        locationId: locationId,
      },
    }))

    // Clear error when user selects location
    if (fieldErrors[`${productId}_locationId`]) {
      setFieldErrors((prev) => ({
        ...prev,
        [`${productId}_locationId`]: undefined,
      }))
    }
  }

  const handleFileUpload = async (file, productId, fieldName) => {
    if (!file) return

    setUploadingFiles((prev) => ({
      ...prev,
      [`${productId}_${fieldName}`]: true,
    }))

    try {
      const response = await uploadImageApi(file)

      if (response.status && response.items?.fileUrl) {
        handleProductFieldChange(productId, fieldName, response.items.fileUrl)
        handleProductFieldChange(productId, `${fieldName}_filename`, file.name)

        // Make AI button visible for ANY file upload (images or documents)
        setAiButtonVisible((prev) => ({
          ...prev,
          [productId]: true,
        }))
      } else {
        console.error("File upload failed:", response)
        showSnackbar("File upload failed", "error")
      }
    } catch (error) {
      console.error("Error uploading file:", error)
      showSnackbar("Error uploading file", "error")
    } finally {
      setUploadingFiles((prev) => ({
        ...prev,
        [`${productId}_${fieldName}`]: false,
      }))
    }
  }

  const handleMultiFileUpload = async (files, productId, fieldName) => {
    if (!files || files.length === 0) return

    // Initialize upload progress indicators for each file
    const uploadItems = files.map((file) => ({
      filename: file.name,
      size: file.size,
      progress: 0,
      status: "uploading",
    }))

    setUploadingFiles((prev) => ({
      ...prev,
      [`${productId}_${fieldName}_items`]: uploadItems,
    }))

    try {
      const formData = new FormData()
      files.forEach((file) => {
        formData.append("images", file)
      })

      const response = await uploadMultiImageApi(formData)

      if (response.status && response.items?.files) {
        // Update progress to 100% for all files
        setUploadingFiles((prev) => ({
          ...prev,
          [`${productId}_${fieldName}_items`]: prev[`${productId}_${fieldName}_items`].map((item) => ({
            ...item,
            progress: 100,
            status: "completed",
          })),
        }))

        // Get existing files if any
        const existingFiles = Array.isArray(productFields[productId]?.[fieldName])
          ? [...productFields[productId][fieldName]]
          : []

        // Combine with new files
        const updatedFiles = [...existingFiles, ...response.items.files].flat()

        // Store the file URLs array
        handleProductFieldChange(productId, fieldName, updatedFiles)

        // Get existing filenames if any
        const existingFilenames = Array.isArray(productFields[productId]?.[`${fieldName}_filenames`])
          ? [...productFields[productId][`${fieldName}_filenames`]]
          : []

        // Store filenames for display purposes - combine with existing
        const filenames = [...existingFilenames, ...files.map((file) => file.name)]
        handleProductFieldChange(productId, `${fieldName}_filenames`, filenames)

        // Make AI button visible for ANY file upload (images or documents)
        setAiButtonVisible((prev) => ({
          ...prev,
          [productId]: true,
        }))

        // Clear the temporary uploading files state after a delay
        setTimeout(() => {
          setUploadingFiles((prev) => {
            const newState = { ...prev }
            delete newState[`${productId}_${fieldName}_items`]
            return newState
          })
        }, 2000)
      } else {
        console.error("Multi-file upload failed:", response)
        showSnackbar("Multi-file upload failed", "error")

        // Update status to show error
        setUploadingFiles((prev) => ({
          ...prev,
          [`${productId}_${fieldName}_items`]: prev[`${productId}_${fieldName}_items`].map((item) => ({
            ...item,
            status: "error",
          })),
        }))
      }
    } catch (error) {
      console.error("Error uploading multiple files:", error)
      showSnackbar("Error uploading multiple files", "error")

      // Update status to show error
      setUploadingFiles((prev) => ({
        ...prev,
        [`${productId}_${fieldName}_items`]:
          prev[`${productId}_${fieldName}_items`]?.map((item) => ({
            ...item,
            status: "error",
          })) || [],
      }))
    }
  }

  const handleRemoveFile = (productId, fieldName, index) => {
    // For single file upload
    if (index === undefined) {
      handleProductFieldChange(productId, fieldName, "")
      handleProductFieldChange(productId, `${fieldName}_filename`, "")

      // Check if we should hide AI button after removing file
      updateAiButtonVisibility(productId)
      return
    }

    // For multi-file upload
    const newFiles = [...productFields[productId][fieldName]]
    newFiles.splice(index, 1)

    // Update the files array
    handleProductFieldChange(productId, fieldName, newFiles)

    // Update filenames array if exists
    if (productFields[productId]?.[`${fieldName}_filenames`]) {
      const newFilenames = [...productFields[productId][`${fieldName}_filenames`]]
      newFilenames.splice(index, 1)
      handleProductFieldChange(productId, `${fieldName}_filenames`, newFilenames)

    // Check if we should hide AI button after removing file
    updateAiButtonVisibility(productId)
  }

  // Helper function to update AI button visibility based on ANY files

    const currentProduct = partnerProducts.find((p) => p._id === productId)
    if (!currentProduct?.initFields?.fields) return

    let hasAnyFiles = false

    // Check all file fields for ANY files
    currentProduct.initFields.fields.forEach((field) => {
      if (field.dataType !== "string" && field.dataType !== "textarea") {
        const fieldValue = productFields[productId]?.[field.fieldName]

        if (fieldValue) {
          if (Array.isArray(fieldValue)) {
            // Multi-file upload - check if array has any files
            hasAnyFiles = hasAnyFiles || fieldValue.length > 0
          } else {
            // Single file upload - check if field has value
            hasAnyFiles = hasAnyFiles || Boolean(fieldValue)
          }
        }
      }
    })

    setAiButtonVisible((prev) => ({
      ...prev,
      [productId]: hasAnyFiles,
    }))
  }

  const runAIDataExtraction = async (productId) => {
    setAiProcessing((prev) => ({
      ...prev,
      [productId]: true,
    }))

    try {
      // Find the current product
      const currentProduct = partnerProducts[0]?.productForm?.find((p) => p._id === productId)

      if (!currentProduct) {
        console.error("Product information not found")
        return
      }

      // Get ALL file URLs for this product (images and documents)
      const fileUrls = []

      if (currentProduct?.initFields?.fields) {
        currentProduct.initFields.fields.forEach((field) => {
          if (
            field.dataType !== "string" &&
            field.dataType !== "textarea" &&
            productFields[productId]?.[field.fieldName]
          ) {
            const fieldValue = productFields[productId][field.fieldName]

            if (Array.isArray(fieldValue)) {
              // Multi-file upload - include all files
              fileUrls.push(...fieldValue)
            } else {
              // Single file upload - include the file
              fileUrls.push(fieldValue)
            }
          }
        })
      }

      // Proceed if we have any files
      if (fileUrls.length === 0) {
        console.warn("No files found for AI processing")
        showSnackbar("No files found for AI processing", "warning")
        return
      }

      const flatFileUrls = fileUrls.flat()

      const payload = {
        reqId: requestId,
        userProductId: currentProduct.userProductId,
        doc: flatFileUrls,
      }
      console.log("AI Extraction Payload:", payload)

      const response = await getAiDataAPI(payload)
      console.log('ai data',response);

      if (response.status && response.items) {
        // Update the form fields with the extracted data
        const extractedData = response.items

        // Create updated product fields
        const updatedFields = {
          ...productFields[productId],
        }

        // Loop through all the extracted fields and update corresponding form fields
        Object.keys(extractedData).forEach((fieldName) => {
          // Only update if the field has a value
          if (extractedData[fieldName] !== null) {
            updatedFields[fieldName] = extractedData[fieldName]
          }
        })

        // Update the product fields state with new data
        setProductFields((prev) => ({
          ...prev,
          [productId]: updatedFields,
        }))

        // Store the AI response data for this product
        setAiResponseData((prev) => ({
          ...prev,
          [productId]: {
            reqId: currentProduct.userProductId,
            userProductId: currentProduct.userProductId,
            doc: fileUrls,
          },
        }))

        showSnackbar("AI data extraction completed successfully", "success")
      } else {
        const errorMessage = response.message || "Failed to extract data from files"
        console.error("AI Extraction Error:", errorMessage)
        showSnackbar(errorMessage, "error")
      }
    } catch (error) {
      console.error("Error in AI data extraction:", error)
      showSnackbar("Error processing files", "error")
    } finally {
      setAiProcessing((prev) => ({
        ...prev,
        [productId]: false,
      }))
    }
  }

  const validateForm = () => {
    const errors = {}
    let isValid = true
    let hasAnyVisibleProduct = false
    const errorMessages = []

    // Validate each product's fields
    Object.entries(formVisibility).forEach(([productId, isVisible]) => {
      // Only validate visible/selected products
      if (!isVisible) return

      hasAnyVisibleProduct = true

      // Find the product to access its field definitions
      const product = partnerProducts.find((p) => p._id === productId)
      if (!product) return

      // Check if product exists in productFields
      if (!productFields[productId]) {
        errors[`${productId}_general`] = "Product data is missing"
        errorMessages.push(`${product.productName}: Product data is missing`)
        isValid = false
        return
      }

      const fields = productFields[productId] || {}

      // Validate location field (required for all products)
      if (!fields.locationId) {
        errors[`${productId}_locationId`] = "Location is required"
        errorMessages.push(`${product.productName}: Location is required`)
        isValid = false
      }

      // Validate each custom field based on the product's initFields
      product?.initFields?.fields?.forEach((field) => {
        const fieldValue = fields[field.fieldName]

        // Only validate fields with dataType "string"
        if (field.dataType === "string") {
          // Check if field is required (all string fields are required)
          if (fieldValue === undefined || fieldValue === null || fieldValue === "") {
            errors[`${productId}_${field.fieldName}`] = `${field.fieldName} is required`
            errorMessages.push(`${product.productName}: ${field.fieldName} is required`)
            isValid = false
          }
        }
      })
    })

    // Check if no products are selected/visible
    if (!hasAnyVisibleProduct) {
      errorMessages.push("Please enable at least one product")
      isValid = false
    }

    // Set field errors to update UI
    setFieldErrors(errors)

    // Show validation errors in snackbar if any
    if (errorMessages.length > 0) {
      showSnackbar(errorMessages[0], "error")
    }

    return isValid
  }

  const handleFormSubmit = async (e) => {
    e.preventDefault()

    // Run validation first
    const isValid = validateForm()
    if (!isValid) return

    // Prepare data for submission with correct structure
    // Only include products that are visible/toggled on
    const jobProductList = Object.entries(formVisibility)
      .filter(([productId, isVisible]) => isVisible)
      .map(([productId]) => {

                                                  productFields[product._id]?.[`${field.fieldName}_filenames`]?.[
                                                    index
                                                  ] || fileUrl.split("/").pop()
                                                const isImage = isImageFile(filename)

                                                return (
                                                  <Box
                                                    key={`uploaded-${index}`}
                                                    sx={{
                                                      p: 2,
                                                      bgcolor: "white",
                                                      borderRadius: 2,
                                                      border: "1px solid #e5e7eb",
                                                      display: "flex",
                                                      alignItems: "center",
                                                      justifyContent: "space-between",
                                                    }}
                                                  >
                                                    <Box sx={{ display: "flex", alignItems: "center" }}>
                                                      <Box
                                                        sx={{
                                                          width: 32,
                                                          height: 32,
                                                          borderRadius: 1.5,
                                                          bgcolor: isImage ? "#f59e0b" : "#10b981",
                                                          display: "flex",
                                                          alignItems: "center",
                                                          justifyContent: "center",
                                                          mr: 2,
                                                        }}
                                                      >
                                                        {isImage ? (
                                                          <ImageOutlined sx={{ color: "white", fontSize: 16 }} />
                                                        ) : (
                                                          <DescriptionOutlined sx={{ color: "white", fontSize: 16 }} />
                                                        )}
                                                      </Box>
                                                      <Box>
                                                        <Typography variant="body2" fontWeight={500}>
                                                          {filename}
                                                        </Typography>
                                                        <Typography variant="caption" color="text.secondary">
                                                          {isImage ? "Image file" : "Document file"}
                                                        </Typography>
                                                      </Box>
                                                    </Box>
                                                    <IconButton
                                                      size="small"
                                                      onClick={(e) => {
                                                        e.stopPropagation()
                                                        handleRemoveFile(product._id, field.fieldName, index)
                                                      }}
                                                      sx={{ color: "#ef4444" }}
                                                    >
                                                      <DeleteOutlined />
                                                    </IconButton>
                                                  </Box>
                                                )
                                              })}
                                          </Stack>

                                          <input
                                            id={`multi-file-input-${fieldKey}`}
                                            type="file"
                                            multiple
                                            style={{ display: "none" }}
                                            onChange={(e) => {
                                              if (e.target.files?.length) {
                                                const filesArray = Array.from(e.target.files)
                                                handleMultiFileUpload(filesArray, product._id, field.fieldName)
                                              }
                                            }}
                                          />
                                        </Box>
                                        {hasError && (
                                          <Typography color="error" variant="caption" sx={{ mt: 1, display: "block" }}>
                                            {errorMessage}
                                          </Typography>
                                        )}
                                      </Box>
                                    ) : null}
                                  </Grid>
                                )
                              })}
                            </Grid>
                          </Box>
                        )}
                      </Card>
                    )
                  })}
                </Stack>
              </CardContent>
            </Card>
          )}

          {/* Action Buttons */}
          <Box
            sx={{
              display: "flex",
              justifyContent: "flex-end",
              gap: 2,
              pt: 2,
              borderTop: "1px solid #e5e7eb",
            }}
          >
            <Button
              variant="outlined"
              onClick={handleCloseAddModal}
              sx={{
                borderColor: "#d1d5db",
                color: "#6b7280",
                "&:hover": {
                  borderColor: "#9ca3af",
                  bgcolor: "#f9fafb",
                },
                borderRadius: 2,
                textTransform: "none",
                fontWeight: 500,
                px: 4,
                py: 1.5,
              }}
            >
              Cancel
            </Button>
            <Button
              variant="contained"
              onClick={handleFormSubmit}
              sx={{
                bgcolor: "#6366f1",
                "&:hover": {
                  bgcolor: "#5b21b6",
                },
                borderRadius: 2,
                textTransform: "none",
                fontWeight: 500,
                px: 4,
                py: 1.5,
                boxShadow: "0 4px 12px rgba(99, 102, 241, 0.3)",
              }}
            >
              Create Case
            </Button>
          </Box>
        </Box>
      </Modal>

      {/* Snackbar for notifications */}
      <Snackbar
        open={snackbar.open}
        autoHideDuration={6000}
        onClose={() => setSnackbar({ ...snackbar, open: false })}
        anchorOrigin={{ vertical: "top", horizontal: "center" }}
      >
        <Alert
          onClose={() => setSnackbar({ ...snackbar, open: false })}
          variant="filled"
          severity={snackbar.severity}
          sx={{ width: "100%" }}
        >
          {snackbar.message}
        </Alert>
      </Snackbar>
    </>
  )
}
