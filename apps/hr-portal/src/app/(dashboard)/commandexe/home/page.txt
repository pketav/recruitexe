const StatusCard = ({ title, value, color, icon }) => {
  // Determine value text color based on status
  const valueColor =
    color === 'blue'
      ? 'text-blue-600'
      : color === 'orange'
        ? 'text-orange-500'
        : color === 'green'
          ? 'text-green-600'
          : 'text-gray-800'

  // Background color for the SVG rectangle
  const bgColor =
    color === 'blue' ? '#5035FE' : color === 'orange' ? '#FFA500' : color === 'green' ? '#089949' : '#CCCCCC'

  return (
    <div className='flex bg-white rounded-lg shadow p-4 h-full items-center'>
      <div className='flex-shrink-0 mr-3'>
        <svg width='46' height='46' viewBox='0 0 46 46' fill='none' xmlns='http://www.w3.org/2000/svg'>
          <rect width='46' height='46' rx='6' fill={bgColor} fillOpacity='0.1' />
          <foreignObject width='46' height='46'>
            <div className='flex items-center justify-center h-full'>{icon}</div>
          </foreignObject>
        </svg>
      </div>
      <div className='flex flex-col'>
        <span className='text-sm text-gray-600'>{title}</span>
        <span className={`text-xl font-bold ${valueColor}`}>{value}</span>
      </div>
    </div>
  )
}
